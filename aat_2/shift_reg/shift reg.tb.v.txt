`timescale 1ns/1ps

module tb_usr_4bit;

reg clk, rst;
reg [1:0] s;
reg [3:0] d;
reg msb_in, lsb_in;
wire [3:0] q;

usr_4bit uut (
    .clk(clk),
    .rst(rst),
    .s(s),
    .d(d),
    .msb_in(msb_in),
    .lsb_in(lsb_in),
    .q(q)
);

always #5 clk = ~clk;

initial begin
    clk = 0;
    rst = 1;
    s = 2'b00;
    d = 4'b0000;
    msb_in = 0;
    lsb_in = 0;

    #10 rst = 0;

    s = 2'b11;
    d = 4'b1011;
    #10;

    s = 2'b00;
    #10;

    s = 2'b01;
    msb_in = 1;
    #10;

    msb_in = 0;
    #10;

    s = 2'b10;
    lsb_in = 1;
    #10;

    lsb_in = 0;
    #10;

    #20 $finish;
end

endmodule
