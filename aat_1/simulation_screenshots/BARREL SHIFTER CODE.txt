
`timescale 1ns/1ps

module barrel_shifter (
    input [3:0] A,
    input [1:0] shift,
    input dir,
    output reg [3:0] Y
);

always @(*) begin
    if (dir == 0)
        Y = A << shift;  // Left shift
    else
        Y = A >> shift;  // Right shift
end

endmodule


// TESTBENCH

module barrel_shifter_tb;

reg [3:0] A;
reg [1:0] shift;
reg dir;
wire [3:0] Y;

barrel_shifter dut (
    .A(A),
    .shift(shift),
    .dir(dir),
    .Y(Y)
);

initial begin
    $monitor("Time=%0t A=%b Shift=%d Dir=%b Y=%b",
             $time, A, shift, dir, Y);

    // Left shift tests
    A = 4'b1011;
    dir = 0;

    shift = 0; #10;
    shift = 1; #10;
    shift = 2; #10;
    shift = 3; #10;

    // Right shift tests
    A = 4'b1011;
    dir = 1;

    shift = 0; #10;
    shift = 1; #10;
    shift = 2; #10;
    shift = 3; #10;

    $finish;
end

endmodule
