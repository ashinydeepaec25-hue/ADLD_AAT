
`timescale 1ns/1ps

module random_sequence_generator (
    input clk, reset,
    output reg [3:0] Q
);

wire feedback;

assign feedback = Q[3] ^ Q[2];

always @(posedge clk) begin
    if (reset)
        Q <= 4'b0001;
    else
        Q <= {Q[2:0], feedback};
end

endmodule


// TESTBENCH

module random_sequence_generator_tb;

reg clk, reset;
wire [3:0] Q;

random_sequence_generator dut (
    .clk(clk),
    .reset(reset),
    .Q(Q)
);

always #5 clk = ~clk;

initial begin
    $monitor("Time=%0t Random Output=%b",
             $time, Q);

    clk = 0;
    reset = 1;
    #10;

    reset = 0;
    #160;

    $finish;
end

endmodule
