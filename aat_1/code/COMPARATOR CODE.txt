
`timescale 1ns/1ps

module comparator_4bit (
    input [3:0] A, B,
    output reg A_greater,
    output reg A_equal,
    output reg A_less
);

always @(*) begin
    if (A > B) begin
        A_greater = 1;
        A_equal = 0;
        A_less = 0;
    end
    else if (A == B) begin
        A_greater = 0;
        A_equal = 1;
        A_less = 0;
    end
    else begin
        A_greater = 0;
        A_equal = 0;
        A_less = 1;
    end
end

endmodule


// TESTBENCH

module comparator_4bit_tb;

reg [3:0] A, B;
wire A_greater, A_equal, A_less;

comparator_4bit dut (
    .A(A),
    .B(B),
    .A_greater(A_greater),
    .A_equal(A_equal),
    .A_less(A_less)
);

initial begin
    $monitor("Time=%0t A=%b B=%b Greater=%b Equal=%b Less=%b",
             $time, A, B, A_greater, A_equal, A_less);

    A = 4'b1010; B = 4'b0101; #10;
    A = 4'b0011; B = 4'b1000; #10;
    A = 4'b0110; B = 4'b0110; #10;
    A = 4'b1111; B = 4'b0001; #10;
    A = 4'b0010; B = 4'b0100; #10;

    $finish;
end

endmodule
