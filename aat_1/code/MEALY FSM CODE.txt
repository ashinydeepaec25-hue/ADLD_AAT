
`timescale 1ns/1ps

module mealy_fsm (
    input clk, reset, x,
    output reg z
);

reg [1:0] state;

parameter S0 = 2'b00,
          S1 = 2'b01,
          S2 = 2'b10;

always @(posedge clk) begin
    if (reset)
        state <= S0;
    else begin
        case (state)

            S0: begin
                if (x)
                    state <= S1;
                else
                    state <= S0;
            end

            S1: begin
                if (x)
                    state <= S1;
                else
                    state <= S2;
            end

            S2: begin
                if (x)
                    state <= S1;
                else
                    state <= S0;
            end

            default: state <= S0;

        endcase
    end
end

always @(*) begin
    z = 0;

    case (state)
        S2: begin
            if (x)
                z = 1;
        end
    endcase
end

endmodule


// TESTBENCH

module mealy_fsm_tb;

reg clk, reset, x;
wire z;

mealy_fsm dut (
    .clk(clk),
    .reset(reset),
    .x(x),
    .z(z)
);

always #5 clk = ~clk;

initial begin
    $monitor("Time=%0t X=%b Z=%b",
             $time, x, z);

    clk = 0;
    reset = 1;
    x = 0;
    #10;

    reset = 0;

    x = 1; #10;
    x = 0; #10;
    x = 1; #10;
    x = 0; #10;
    x = 1; #10;
    x = 1; #10;
    x = 0; #10;

    $finish;
end

endmodule
