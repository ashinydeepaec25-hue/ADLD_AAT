
`timescale 1ns/1ps

// 4-BIT SYNCHRONOUS COUNTER

module synchronous_counter (
    input clk, reset,
    output reg [3:0] Q
);

always @(posedge clk) begin
    if (reset)
        Q <= 4'b0000;
    else
        Q <= Q + 1'b1;
end

endmodule


// 4-BIT ASYNCHRONOUS COUNTER

module asynchronous_counter (
    input clk, reset,
    output reg [3:0] Q
);

always @(posedge clk or posedge reset) begin
    if (reset)
        Q[0] <= 0;
    else
        Q[0] <= ~Q[0];
end

always @(negedge Q[0] or posedge reset) begin
    if (reset)
        Q[1] <= 0;
    else
        Q[1] <= ~Q[1];
end

always @(negedge Q[1] or posedge reset) begin
    if (reset)
        Q[2] <= 0;
    else
        Q[2] <= ~Q[2];
end

always @(negedge Q[2] or posedge reset) begin
    if (reset)
        Q[3] <= 0;
    else
        Q[3] <= ~Q[3];
end

endmodule


// COMMON TESTBENCH

module counter_tb;

reg clk, reset;

wire [3:0] sync_Q;
wire [3:0] async_Q;

synchronous_counter sync_counter (
    .clk(clk),
    .reset(reset),
    .Q(sync_Q)
);

asynchronous_counter async_counter (
    .clk(clk),
    .reset(reset),
    .Q(async_Q)
);

always #5 clk = ~clk;

initial begin

    $monitor("Time=%0t Sync_Q=%b Async_Q=%b",
             $time, sync_Q, async_Q);

    clk = 0;
    reset = 1;
    #10;

    reset = 0;
    #200;

    $finish;

end

endmodule
