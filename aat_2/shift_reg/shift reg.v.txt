`timescale 1ns/1ps

module usr_4bit (
    input wire clk,
    input wire rst,
    input wire [1:0] s,
    input wire [3:0] d,
    input wire msb_in,
    input wire lsb_in,
    output reg [3:0] q
);

always @(posedge clk or posedge rst) begin
    if (rst)
        q <= 4'b0000;
    else begin
        case (s)
            2'b00: q <= q;
            2'b01: q <= {msb_in, q[3:1]};
            2'b10: q <= {q[2:0], lsb_in};
            2'b11: q <= d;
            default: q <= q;
        endcase
    end
end

endmodule
