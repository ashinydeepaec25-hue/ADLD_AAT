
`timescale 1ns/1ps

module divide_by_n_counter (
    input clk, reset,
    output reg [3:0] Q
);

always @(posedge clk) begin
    if (reset)
        Q <= 4'b0000;
    else if (Q == 4'd9)
        Q <= 4'b0000;
    else
        Q <= Q + 1'b1;
end

endmodule


// TESTBENCH

module divide_by_n_counter_tb;

reg clk, reset;
wire [3:0] Q;

divide_by_n_counter dut (
    .clk(clk),
    .reset(reset),
    .Q(Q)
);

always #5 clk = ~clk;

initial begin
    $monitor("Time=%0t Q=%d Binary=%b",
             $time, Q, Q);

    clk = 0;
    reset = 1;
    #10;

    reset = 0;
    #120;

    $finish;
end

endmodule
