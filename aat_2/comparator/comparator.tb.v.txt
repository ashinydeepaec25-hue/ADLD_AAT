`timescale 1ns/1ps

module tb_4bit_cmp;

reg [3:0] A;
reg [3:0] B;

wire A_gt_B;
wire A_lt_B;
wire A_et_B;

cmp_4bit uut (
    .A(A),
    .B(B),
    .A_gt_B(A_gt_B),
    .A_lt_B(A_lt_B),
    .A_et_B(A_et_B)
);

initial begin
    A = 4'd5;
    B = 4'd5;
    #10;

    A = 4'd12;
    B = 4'd3;
    #10;

    A = 4'd2;
    B = 4'd9;
    #10;

    A = 4'd15;
    B = 4'd15;
    #10;

    $finish;
end

endmodule
